#include <Adafruit_GFX.h>
#include <Adafruit_ST7789.h>
#include <SPI.h>

// TFT - połączenia pozostają bez zmian.
constexpr int8_t TFT_SCLK = D8;   // GPIO8
constexpr int8_t TFT_MOSI = D10;  // GPIO10
constexpr int8_t TFT_CS = -1;     // CS fizycznie połączone z GND
constexpr int8_t TFT_DC = D6;     // GPIO11
constexpr int8_t TFT_RST = D2;    // GPIO25
constexpr int8_t TFT_BL = D7;     // GPIO12

// Kierunki przycisków po obróceniu obrazu. Lutowanie pozostaje bez zmian.
constexpr uint8_t BTN_UP = D0;     // GPIO1
constexpr uint8_t BTN_DOWN = D5;   // GPIO24
constexpr uint8_t BTN_LEFT = D1;   // GPIO0
constexpr uint8_t BTN_RIGHT = D4;  // GPIO23

constexpr uint32_t TFT_SPI_HZ = 4000000;
constexpr uint32_t DEBOUNCE_MS = 30;

constexpr uint16_t COLOR_BG = 0x0841;
constexpr uint16_t COLOR_SURFACE = 0x10C3;
constexpr uint16_t COLOR_SURFACE_2 = 0x2165;
constexpr uint16_t COLOR_ACCENT = 0x4E5F;
constexpr uint16_t COLOR_TEXT = ST77XX_WHITE;
constexpr uint16_t COLOR_MUTED = 0x9CF3;
constexpr uint16_t COLOR_OK = 0x4E8D;

Adafruit_ST7789 tft(&SPI, TFT_CS, TFT_DC, TFT_RST);

enum Direction : uint8_t { UP, DOWN, LEFT, RIGHT, DIRECTION_COUNT };
enum Page : uint8_t { MENU, DISPLAY_TEST, BUTTON_TEST, SYSTEM_TEST };

struct ButtonState {
  uint8_t pin;
  const char *name;
  const char *pinName;
  bool rawState;
  bool stableState;
  bool seen;
  uint32_t changedAt;
  uint32_t presses;
};

ButtonState buttons[DIRECTION_COUNT] = {
    {BTN_UP, "UP", "D0", HIGH, HIGH, false, 0, 0},
    {BTN_DOWN, "DOWN", "D5", HIGH, HIGH, false, 0, 0},
    {BTN_LEFT, "LEFT", "D1", HIGH, HIGH, false, 0, 0},
    {BTN_RIGHT, "RIGHT", "D4", HIGH, HIGH, false, 0, 0},
};

const char *menuItems[] = {"TEST EKRANU", "PRZYCISKI", "SYSTEM"};
constexpr uint8_t MENU_ITEM_COUNT = sizeof(menuItems) / sizeof(menuItems[0]);

Page currentPage = MENU;
uint8_t selectedItem = 0;
uint8_t displayStage = 0;
int8_t lastDirection = -1;
int16_t scannerX = 20;
int16_t previousScannerX = 20;
uint32_t scannerUpdatedAt = 0;
uint32_t systemUpdatedAt = 0;

void drawCenteredText(const char *text, int16_t centerX, int16_t y,
                      uint8_t size, uint16_t color) {
  int16_t x1, y1;
  uint16_t width, height;
  tft.setTextSize(size);
  tft.setTextColor(color);
  tft.getTextBounds(text, 0, y, &x1, &y1, &width, &height);
  tft.setCursor(centerX - static_cast<int16_t>(width) / 2, y);
  tft.print(text);
}

void drawHeader(const char *title, const char *subtitle) {
  tft.fillRect(0, 0, 240, 39, COLOR_BG);
  tft.fillRoundRect(10, 10, 9, 9, 3, COLOR_ACCENT);
  tft.setTextSize(2);
  tft.setTextColor(COLOR_TEXT);
  tft.setCursor(27, 7);
  tft.print(title);
  tft.setTextSize(1);
  tft.setTextColor(COLOR_MUTED);
  tft.setCursor(28, 26);
  tft.print(subtitle);
  tft.drawFastHLine(10, 38, 220, COLOR_SURFACE_2);
}

void drawFooter(const char *leftText, const char *rightText) {
  tft.fillRect(0, 211, 240, 29, COLOR_BG);
  tft.drawFastHLine(10, 211, 220, COLOR_SURFACE_2);
  tft.setTextSize(1);
  tft.setTextColor(COLOR_MUTED);
  tft.setCursor(10, 222);
  tft.print(leftText);

  int16_t x1, y1;
  uint16_t width, height;
  tft.getTextBounds(rightText, 0, 222, &x1, &y1, &width, &height);
  tft.setCursor(230 - width, 222);
  tft.print(rightText);
}

uint8_t seenButtonCount() {
  uint8_t count = 0;
  for (const ButtonState &button : buttons) {
    if (button.seen) ++count;
  }
  return count;
}

void drawMenu() {
  tft.fillScreen(COLOR_BG);
  drawHeader("BUDDY", "HARDWARE LAB");

  for (uint8_t i = 0; i < MENU_ITEM_COUNT; ++i) {
    const int16_t y = 49 + i * 48;
    const bool selected = i == selectedItem;
    tft.fillRoundRect(12, y, 216, 39, 7,
                      selected ? COLOR_SURFACE_2 : COLOR_SURFACE);
    if (selected) {
      tft.fillRoundRect(12, y, 5, 39, 3, COLOR_ACCENT);
      tft.drawRoundRect(12, y, 216, 39, 7, COLOR_ACCENT);
    }

    tft.setTextSize(1);
    tft.setTextColor(selected ? COLOR_ACCENT : COLOR_MUTED);
    tft.setCursor(25, y + 16);
    tft.print('0');
    tft.print(i + 1);

    tft.setTextColor(COLOR_TEXT);
    tft.setCursor(54, y + 16);
    tft.print(menuItems[i]);

    tft.setTextColor(selected ? COLOR_ACCENT : COLOR_MUTED);
    tft.setCursor(211, y + 16);
    tft.print('>');
  }

  tft.fillRect(12, 194, 216, 12, COLOR_BG);
  tft.setTextSize(1);
  tft.setTextColor(COLOR_MUTED);
  tft.setCursor(13, 195);
  if (lastDirection < 0) {
    tft.print("GOTOWY - WYBIERZ TEST");
  } else {
    tft.print("OSTATNI: ");
    tft.setTextColor(COLOR_ACCENT);
    tft.print(buttons[lastDirection].name);
    tft.setTextColor(COLOR_MUTED);
    tft.print("  KLAWISZE ");
    tft.print(seenButtonCount());
    tft.print("/4");
  }
  drawFooter("^ v  WYBOR", ">  WEJDZ");
}

uint16_t gray565(uint8_t level) {
  return ((level & 0xF8) << 8) | ((level & 0xFC) << 3) | (level >> 3);
}

  drawCenteredText("8 POZIOMOW SZAROSCI", 120, 184, 1, COLOR_TEXT);

void drawMotionTestBase() {
  tft.fillRect(0, 39, 240, 172, COLOR_BG);
  for (int16_t x = 20; x <= 220; x += 20) {
    tft.drawFastVLine(x, 51, 137, COLOR_SURFACE_2);
  }
  for (int16_t y = 51; y <= 188; y += 20) {
    tft.drawFastHLine(20, y, 201, COLOR_SURFACE_2);
  }
  tft.drawRect(19, 50, 203, 139, COLOR_MUTED);
  drawCenteredText("TEST ODSWIEZANIA", 120, 199, 1, COLOR_MUTED);
  scannerX = previousScannerX = 20;
  tft.drawFastVLine(scannerX, 51, 137, COLOR_ACCENT);
  scannerUpdatedAt = millis();
}

void drawDisplayTest() {
  tft.fillScreen(COLOR_BG);
  char subtitle[24];
  snprintf(subtitle, sizeof(subtitle), "ETAP %u / 3", displayStage + 1);
  drawHeader("EKRAN", subtitle);

  if (displayStage == 0) {
    drawGeometryTest();
  } else if (displayStage == 1) {
    drawGrayscaleTest();
  } else {
    drawMotionTestBase();
  }
  drawFooter("<  BACK", ">  DALEJ");
}

void restoreGridColumn(int16_t x) {
  if ((x - 20) % 20 == 0) {
    tft.drawFastVLine(x, 51, 137, COLOR_SURFACE_2);
  } else {
    tft.drawFastVLine(x, 51, 137, COLOR_BG);
    for (int16_t y = 51; y <= 188; y += 20) {
      tft.drawPixel(x, y, COLOR_SURFACE_2);
    }
  }
}

void updateMotionTest() {
  if (currentPage != DISPLAY_TEST || displayStage != 2) return;
  if (millis() - scannerUpdatedAt < 35) return;

  restoreGridColumn(previousScannerX);
  scannerX += 2;
  if (scannerX > 220) scannerX = 20;
  tft.drawFastVLine(scannerX, 51, 137, COLOR_ACCENT);
  previousScannerX = scannerX;
  scannerUpdatedAt = millis();
}

void drawButtonCard(Direction direction, int16_t x, int16_t y, int16_t w,
                    int16_t h) {
  const ButtonState &button = buttons[direction];
  const bool recent = lastDirection == static_cast<int8_t>(direction);
  const uint16_t border = recent ? COLOR_ACCENT
                                 : (button.seen ? COLOR_OK : COLOR_SURFACE_2);
  tft.fillRoundRect(x, y, w, h, 7, COLOR_SURFACE);
  tft.drawRoundRect(x, y, w, h, 7, border);
  if (recent) tft.drawRoundRect(x + 2, y + 2, w - 4, h - 4, 6, border);

  drawCenteredText(button.name, x + w / 2, y + 8, 1, COLOR_TEXT);
  char count[12];
  snprintf(count, sizeof(count), "%s  %lu", button.pinName,
           static_cast<unsigned long>(button.presses));
  drawCenteredText(count, x + w / 2, y + 24, 1,
                   button.seen ? border : COLOR_MUTED);
}

void drawButtonTest() {
  tft.fillScreen(COLOR_BG);
  drawHeader("PRZYCISKI", "LICZNIK I KIERUNEK");

  drawButtonCard(UP, 84, 48, 72, 42);
  drawButtonCard(LEFT, 8, 96, 72, 42);
  drawButtonCard(RIGHT, 160, 96, 72, 42);
  drawButtonCard(DOWN, 84, 144, 72, 42);

  char summary[28];
  snprintf(summary, sizeof(summary), "WYKRYTO %u / 4 KLAWISZE",
           seenButtonCount());
  drawCenteredText(summary, 120, 197, 1,
                   seenButtonCount() == 4 ? COLOR_OK : COLOR_MUTED);
  drawFooter("<  BACK", ">  SYSTEM");
}

void updateSystemUptime() {
  if (currentPage != SYSTEM_TEST) return;
  if (millis() - systemUpdatedAt < 1000) return;
  systemUpdatedAt = millis();

  tft.fillRect(130, 193, 98, 15, COLOR_BG);
  tft.setTextSize(1);
  tft.setTextColor(COLOR_TEXT);
  tft.setCursor(167, 196);
  tft.print(millis() / 1000);
  tft.print(" s");
}

void initializeButtons() {
  for (ButtonState &button : buttons) {
    pinMode(button.pin, INPUT_PULLUP);
    const bool state = digitalRead(button.pin);
    button.rawState = state;
    button.stableState = state;
    button.changedAt = millis();
  }
}

void initializeDisplayMode3() {
  SPI.begin(TFT_SCLK, -1, TFT_MOSI, -1);
  tft.init(240, 240, SPI_MODE3);
  tft.setSPISpeed(TFT_SPI_HZ);

  digitalWrite(TFT_RST, HIGH);
  delay(100);
  digitalWrite(TFT_RST, LOW);
  delay(100);
  digitalWrite(TFT_RST, HIGH);
  delay(200);

  tft.sendCommand(ST77XX_SWRESET);
  delay(150);
  tft.sendCommand(ST77XX_SLPOUT);
  delay(120);

  const uint8_t madctl[] = {0x00};
  tft.sendCommand(ST77XX_MADCTL, madctl, sizeof(madctl));
  const uint8_t ramctrl[] = {0x00, 0xE0};
  tft.sendCommand(0xB0, ramctrl, sizeof(ramctrl));
  const uint8_t colmod[] = {0x55};
  tft.sendCommand(ST77XX_COLMOD, colmod, sizeof(colmod));
  delay(10);
  const uint8_t porctrl[] = {0x0C, 0x0C, 0x00, 0x33, 0x33};
  tft.sendCommand(0xB2, porctrl, sizeof(porctrl));
  const uint8_t gctrl[] = {0x35};
  tft.sendCommand(0xB7, gctrl, sizeof(gctrl));
  const uint8_t vcoms[] = {0x28};
  tft.sendCommand(0xBB, vcoms, sizeof(vcoms));
  const uint8_t lcmctrl[] = {0x2C};
  tft.sendCommand(0xC0, lcmctrl, sizeof(lcmctrl));
  const uint8_t vdvvrhen[] = {0x01};
  tft.sendCommand(0xC2, vdvvrhen, sizeof(vdvvrhen));
  const uint8_t vrhs[] = {0x10};
  tft.sendCommand(0xC3, vrhs, sizeof(vrhs));
  const uint8_t vdvs[] = {0x20};
  tft.sendCommand(0xC4, vdvs, sizeof(vdvs));
  const uint8_t frctrl2[] = {0x0F};
  tft.sendCommand(0xC6, frctrl2, sizeof(frctrl2));
  const uint8_t pwctrl1[] = {0xA4, 0xA1};
  tft.sendCommand(0xD0, pwctrl1, sizeof(pwctrl1));
  const uint8_t positiveGamma[] = {0xD0, 0x00, 0x02, 0x07, 0x0A, 0x28, 0x32,
                                   0x44, 0x42, 0x06, 0x0E, 0x12, 0x14, 0x17};
  tft.sendCommand(0xE0, positiveGamma, sizeof(positiveGamma));
  const uint8_t negativeGamma[] = {0xD0, 0x00, 0x02, 0x07, 0x0A, 0x28, 0x31,
                                   0x54, 0x47, 0x0E, 0x1C, 0x17, 0x1B, 0x1E};
  tft.sendCommand(0xE1, negativeGamma, sizeof(negativeGamma));
  tft.sendCommand(ST77XX_INVON);
  tft.sendCommand(ST77XX_NORON);
  delay(10);
  tft.sendCommand(ST77XX_DISPON);
  delay(120);

  tft.setRotation(2);
  tft.setTextWrap(false);
}

void openSelectedPage() {
  if (selectedItem == 0) {
    currentPage = DISPLAY_TEST;
    displayStage = 0;
    drawDisplayTest();
  } else if (selectedItem == 1) {
    currentPage = BUTTON_TEST;
    drawButtonTest();
  } else {
    currentPage = SYSTEM_TEST;
    drawSystemTest();
  }
}

void handlePress(Direction direction) {
  ButtonState &button = buttons[direction];
  button.presses++;
  button.seen = true;
  lastDirection = direction;

  Serial.print("Przycisk ");
  Serial.print(button.name);
  Serial.print(" (");
  Serial.print(button.pinName);
  Serial.println(") OK");

  if (currentPage == MENU) {
    if (direction == UP) {
      selectedItem = (selectedItem + MENU_ITEM_COUNT - 1) % MENU_ITEM_COUNT;
    } else if (direction == DOWN) {
      selectedItem = (selectedItem + 1) % MENU_ITEM_COUNT;
    } else if (direction == RIGHT) {
      openSelectedPage();
      return;
    }
    drawMenu();
    return;
  }

  if (currentPage == DISPLAY_TEST) {
    if (direction == LEFT) {
      currentPage = MENU;
      drawMenu();
    } else if (direction == RIGHT) {
      displayStage = (displayStage + 1) % 3;
      drawDisplayTest();
    }
    return;
  }

  if (currentPage == BUTTON_TEST) {
    drawButtonTest();
    if (direction == LEFT) {
      delay(120);
      currentPage = MENU;
      drawMenu();
    } else if (direction == RIGHT) {
      delay(120);
      currentPage = SYSTEM_TEST;
      drawSystemTest();
    }
    return;
  }

  if (currentPage == SYSTEM_TEST) {
    if (direction == LEFT) {
      currentPage = MENU;
      drawMenu();
    } else if (direction == RIGHT) {
      runBootVisualTest();
      drawSystemTest();
    }
  }
}

void pollButtons() {
  const uint32_t now = millis();
  for (uint8_t index = 0; index < DIRECTION_COUNT; ++index) {
    ButtonState &button = buttons[index];
    const bool sample = digitalRead(button.pin);

    if (sample != button.rawState) {
      button.rawState = sample;
      button.changedAt = now;
    }

    if ((now - button.changedAt >= DEBOUNCE_MS) &&
        (button.stableState != button.rawState)) {
      button.stableState = button.rawState;
      if (button.stableState == LOW) {
        handlePress(static_cast<Direction>(index));
      }
    }
  }
}

void setup() {
  Serial.begin(115200);
  delay(250);
  initializeButtons();

  pinMode(TFT_BL, OUTPUT);
  digitalWrite(TFT_BL, HIGH);
  initializeDisplayMode3();

  Serial.println();
  Serial.println("Buddy hardware lab started");
  Serial.println("TFT: rotation 180, SPI MODE3 at 4 MHz, CS tied to GND");
  Serial.println("Controls: UP/DOWN select, RIGHT enter, LEFT back");
  Serial.println("Buzzer: disabled / D9 untouched");

  runBootVisualTest();
  drawMenu();
}

void loop() {
  pollButtons();
  updateMotionTest();
  updateSystemUptime();
  delay(2);
}
